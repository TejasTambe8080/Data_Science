📊 PART 2 — BASIC PLOTLY CHARTS
🔹 Install (if needed)
pip install plotly
🔹 Line Chart
import plotly.express as px

x = [1,2,3]
y = [10,20,30]

fig = px.line(x=x, y=y)
fig.show()
🔹 Bar Chart
names = ["IT", "HR"]
values = [60000, 40000]

fig = px.bar(x=names, y=values)
fig.show()
🔹 Scatter Plot
fig = px.scatter(x=[1,2,3], y=[10,20,30])
fig.show()
💻 PART 3 — USING DATAFRAME
import pandas as pd

df = pd.DataFrame({
    "dept": ["IT", "HR", "IT"],
    "salary": [50000, 40000, 60000]
})

fig = px.bar(df, x="dept", y="salary")
fig.show()
